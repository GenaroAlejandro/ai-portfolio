Stationarity (ADF)¶

In [15]:
def adf_report(series, name="series"):
    res = adfuller(series, autolag='AIC')
    out = {
        'name': name,
        'test_statistic': res[0],
        'p_value': res[1],
        'lags_used': res[2],
        'n_obs': res[3],
        'crit_1%': res[4]['1%'],
        'crit_5%': res[4]['5%'],
        'crit_10%': res[4]['10%'],
    }
    return pd.Series(out)

Make series stationary (p < 0.05): log¶

In [ ]:
series = np.log(df['Passengers'])     # start with log transform (stabilize variance)
max_diff = 5                          # safety cap
best_p = 1.0
best_diff = 0
best_series = series.copy()

for d in range(1, max_diff + 1):
    s_diff = series.diff(d).dropna()
    result = adf_report(s_diff, name=f'log(Passengers) diff({d})')
    p_val = result['p_value']
    print(f"d={d} -> p-value={p_val:.4f}")
    if p_val < 0.05:
        best_p = p_val
        best_diff = d
        best_series = s_diff
        print(f"Stationarity achieved at d={d} (p={p_val:.4f})")
        break
else:
    print("No stationarity reached up to d=5; consider seasonal differencing.")

# Save the final stationary series
s_stationary = best_series

# Plot it
ax = s_stationary.plot(title=f'Stationary Series after {best_diff} difference(s) (p={best_p:.4f})')
ax.set_xlabel('Year'); ax.set_ylabel('Differenced log(Passengers)')
plt.show()

df_log = np.log(df['Passengers'])
train_data = df_log[:'1958']
test_data  = df_log['1959':]

# --- Fit SARIMA ---
# start with (p,d,q)=(1,best_diff,1) and seasonal_order=(1,1,1,12)
sarima_model = sm.tsa.statespace.SARIMAX(
    train_data,
    order=(1, best_diff, 1),
    seasonal_order=(1, 1, 1, 12),
    enforce_stationarity=False,
    enforce_invertibility=False
)
sarima_result = sarima_model.fit()
